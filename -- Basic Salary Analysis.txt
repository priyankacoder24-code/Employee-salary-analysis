-- Basic Salary Analysis
SELECT 
    ROUND(AVG(salary), 2) AS average_salary,
    MAX(salary) AS highest_salary,
    MIN(salary) AS lowest_salary
FROM employees;


-- Department-wise Analysis
SELECT
    department,
    COUNT(*) AS employee_count,
    ROUND(AVG(salary), 2) AS average_salary
FROM employees
GROUP BY department
ORDER BY average_salary DESC;


-- Experience Level Analysis
SELECT
    experience_level,
    COUNT(*) AS employee_count,
    ROUND(AVG(salary), 2) AS average_salary
FROM employees
GROUP BY experience_level
ORDER BY average_salary DESC;


-- Job Role-wise Analysis
SELECT
    job_role,
    COUNT(*) AS employee_count,
    ROUND(AVG(salary), 2) AS average_salary
FROM employees
GROUP BY job_role
ORDER BY average_salary DESC;


-- Top 10 Highest Paid Employees
SELECT
    employee_id,
    job_role,
    department,
    experience_years,
    salary
FROM employees
ORDER BY salary DESC
LIMIT 10;


-- Gender-wise Analysis
SELECT
    gender,
    COUNT(*) AS employee_count,
    ROUND(AVG(salary), 2) AS average_salary
FROM employees
GROUP BY gender
ORDER BY average_salary DESC;


-- Education-wise Analysis
SELECT
    education,
    COUNT(*) AS employee_count,
    ROUND(AVG(salary), 2) AS average_salary
FROM employees
GROUP BY education
ORDER BY average_salary DESC;


-- Salary vs Experience
SELECT
    experience_years,
    COUNT(*) AS employee_count,
    ROUND(AVG(salary), 2) AS average_salary
FROM employees
GROUP BY experience_years
ORDER BY experience_years;
